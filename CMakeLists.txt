cmake_minimum_required(VERSION 3.25)
project(XScarlet VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "XScarlet builds for 64-bit Windows with MSVC")
endif()
include(CTest)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
option(XS_VULKAN "Build the headless Vulkan backend and pixel check (requires Vulkan SDK/loader)" OFF)
option(XS_DXVK "Build Direct3D 11 on Vulkan for the vulkan backend (requires meson and glslang)" OFF)
add_library(xs_options INTERFACE)
target_include_directories(xs_options INTERFACE src)
target_compile_options(xs_options INTERFACE /W4 /permissive-)
set(xs_version_header "${PROJECT_BINARY_DIR}/generated/app/version.hpp")
add_custom_target(xs_version_header
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${PROJECT_SOURCE_DIR}" "-DOUTPUT=${xs_version_header}"
    -DVERSION=${PROJECT_VERSION} -P "${PROJECT_SOURCE_DIR}/cmake/Version.cmake"
  BYPRODUCTS "${xs_version_header}" VERBATIM)
add_library(xs_version INTERFACE)
target_include_directories(xs_version INTERFACE "${PROJECT_BINARY_DIR}/generated")
add_dependencies(xs_version xs_version_header)
if(XS_DXVK)
  include(cmake/Dxvk.cmake)
endif()
add_subdirectory(src)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

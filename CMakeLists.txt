cmake_minimum_required(VERSION 3.25)
project(XScarlet VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(cmake/Dependencies.cmake)
option(XS_SANITIZE "Instrument project code with ASan and UBSan" OFF)
option(XS_GRAPHICS_TEST "Test the desktop window under X11 (xdotool and display required)" OFF)
option(XS_VULKAN "Build the headless Vulkan backend and pixel check (requires Vulkan SDK/loader)" OFF)
add_library(xs_options INTERFACE)
target_include_directories(xs_options INTERFACE src)
if(MSVC)
  target_compile_options(xs_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(xs_options INTERFACE -Wall -Wextra -Wpedantic)
  if(XS_SANITIZE)
    target_compile_options(xs_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(xs_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(xs_pe src/formats/pe/inspect.cpp src/formats/pe/directories.cpp)
target_link_libraries(xs_pe PUBLIC nlohmann_json::nlohmann_json PRIVATE xs_options)
add_library(xs_game_config src/formats/gdk/game_config.cpp)
target_link_libraries(xs_game_config PUBLIC nlohmann_json::nlohmann_json PRIVATE xs_options pugixml::pugixml)
add_library(xs_compat src/compatibility/assessment.cpp src/compatibility/modules.cpp)
target_link_libraries(xs_compat PUBLIC nlohmann_json::nlohmann_json PRIVATE xs_options)
add_library(xs_core src/core/settings.cpp src/core/console.cpp)
target_link_libraries(xs_core PUBLIC spdlog::spdlog PRIVATE xs_options nlohmann_json::nlohmann_json)
if(WIN32)
  set(xs_platform windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(xs_platform linux)
else()
  message(FATAL_ERROR "Host runtime supports Windows and Linux")
endif()
find_package(Threads REQUIRED)
add_library(xs_host STATIC src/platform/host/runtime.cpp src/platform/${xs_platform}/runtime.cpp)
target_include_directories(xs_host PUBLIC src)
target_link_libraries(xs_host PRIVATE xs_options ${CMAKE_DL_LIBS} PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(xs_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
add_library(xs_gl STATIC src/frontend/gl.c)
add_library(xs_graphics STATIC src/graphics/abstraction/clear_pass.cpp)
target_link_libraries(xs_graphics PRIVATE xs_options)
if(XS_VULKAN)
  find_package(Vulkan REQUIRED)
  add_library(xs_vulkan STATIC src/graphics/vulkan/offscreen.cpp)
  target_link_libraries(xs_vulkan PRIVATE xs_options xs_graphics Vulkan::Vulkan)
  add_executable(xscarlet-vk-check src/tools/vulkan_check.cpp)
  target_link_libraries(xscarlet-vk-check PRIVATE xs_options xs_vulkan nlohmann_json::nlohmann_json)
endif()
target_include_directories(xs_gl SYSTEM PUBLIC ${glfw_SOURCE_DIR}/deps)
add_library(xs_xvd src/formats/xvd/identify.cpp)
target_link_libraries(xs_xvd PUBLIC nlohmann_json::nlohmann_json PRIVATE xs_options)
add_library(xs_guest src/guest/image.cpp)
target_link_libraries(xs_guest PUBLIC nlohmann_json::nlohmann_json xs_host PRIVATE xs_options xs_pe)
if(WIN32)
  add_library(xs_runtime src/runtime/runtime.cpp src/runtime/windows/execute.cpp)
  target_compile_definitions(xs_runtime PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
  add_library(xs_runtime src/runtime/runtime.cpp src/runtime/unsupported.cpp)
endif()
target_link_libraries(xs_runtime PUBLIC nlohmann_json::nlohmann_json PRIVATE xs_options xs_pe xs_compat xs_guest)
target_include_directories(xs_runtime PUBLIC src)
add_library(xs_launch src/app/launch.cpp)
target_link_libraries(xs_launch PRIVATE xs_options xs_host xs_pe xs_compat xs_xvd xs_guest xs_runtime)
set(xs_version_header "${CMAKE_CURRENT_BINARY_DIR}/generated/app/version.hpp")
add_custom_target(xs_version_header
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${PROJECT_SOURCE_DIR}" "-DOUTPUT=${xs_version_header}"
    -DVERSION=${PROJECT_VERSION} -P "${PROJECT_SOURCE_DIR}/cmake/Version.cmake"
  BYPRODUCTS "${xs_version_header}" VERBATIM)
add_library(xs_version INTERFACE)
target_include_directories(xs_version INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_dependencies(xs_version xs_version_header)
add_library(xs_frontend src/frontend/desktop.cpp src/frontend/file_dialog.cpp)
target_compile_definitions(xs_frontend PRIVATE GLFW_INCLUDE_NONE)
if(WIN32)
  target_compile_definitions(xs_frontend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
target_link_libraries(xs_frontend PRIVATE xs_options xs_version xs_imgui xs_gl glfw nfd Threads::Threads)
add_executable(xscarlet src/app/main.cpp)
target_link_libraries(xscarlet PRIVATE xs_options xs_version xs_launch xs_frontend xs_pe xs_host xs_game_config xs_compat xs_xvd xs_guest xs_runtime)
if(MINGW)
  target_link_options(xscarlet PRIVATE -municode)
endif()
if(BUILD_TESTING)
  add_library(xs_host_test_module MODULE tests/host_test_module.cpp)
  add_executable(xs_tests tests/tests.cpp tests/pe_directories.cpp tests/host_runtime.cpp tests/evidence.cpp tests/clear_pass.cpp
    tests/scarlett_profile.cpp tests/guest_image.cpp tests/xvd.cpp tests/runtime.cpp)
  target_link_libraries(xs_tests PRIVATE xs_options xs_launch xs_pe xs_core xs_catch xs_host xs_game_config xs_compat xs_graphics xs_xvd xs_guest xs_runtime)
  add_dependencies(xs_tests xs_host_test_module)
  target_compile_definitions(xs_tests PRIVATE "XS_TEST_MODULE=\"$<TARGET_FILE:xs_host_test_module>\"")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/host-input.bin" CONTENT "host runtime")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/host-empty.bin" CONTENT "")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(fixture "${CMAKE_CURRENT_BINARY_DIR}/synthetic.pe")
  add_test(NAME fixture COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tests/corpus/generate.py" "${fixture}")
  set_tests_properties(fixture PROPERTIES FIXTURES_SETUP pe_file)
  foreach(variant scarlett-modules delay)
    add_test(NAME fixture_${variant} COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tests/corpus/generate.py"
      "${CMAKE_CURRENT_BINARY_DIR}/${variant}.pe" ${variant})
    set_tests_properties(fixture_${variant} PROPERTIES FIXTURES_SETUP pe_file)
  endforeach()
  add_test(NAME units COMMAND xs_tests "~[host]")
  set_tests_properties(units PROPERTIES FIXTURES_REQUIRED pe_file)
  add_test(NAME host_runtime COMMAND xs_tests "[host]")
  set_tests_properties(host_runtime PROPERTIES TIMEOUT 20)
  add_test(NAME inspect COMMAND xscarlet inspect "${fixture}")
  set_tests_properties(inspect PROPERTIES FIXTURES_REQUIRED pe_file PASS_REGULAR_EXPRESSION "PE32\\+")
  add_test(NAME assess_cli COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tests/assess_cli.py"
    $<TARGET_FILE:xscarlet> "${fixture}" "${PROJECT_SOURCE_DIR}/tests/corpus/game.config")
  set_tests_properties(assess_cli PROPERTIES FIXTURES_REQUIRED pe_file)
  if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_subdirectory(tests/guest)
    add_test(NAME guest_execution COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tests/guest/run_guests.py"
      $<TARGET_FILE:xscarlet> $<TARGET_FILE:xs_guest_raw> $<TARGET_FILE:xs_guest_crt>)
    set_tests_properties(guest_execution PROPERTIES TIMEOUT 60)
  endif()
  add_test(NAME version COMMAND xscarlet --version)
  set_tests_properties(version PROPERTIES PASS_REGULAR_EXPRESSION "^XScarlet v[0-9]+\\.[0-9]+\\.[0-9]+")
  if(XS_GRAPHICS_TEST)
    find_program(XDOTOOL xdotool REQUIRED)
    add_test(NAME desktop_window COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tests/launch_window.py"
      $<TARGET_FILE:xscarlet> "${fixture}" "${XDOTOOL}")
    set_tests_properties(desktop_window PROPERTIES TIMEOUT 60 FIXTURES_REQUIRED pe_file)
  endif()
  if(XS_VULKAN)
    add_test(NAME vulkan_readback COMMAND xscarlet-vk-check)
    set_tests_properties(vulkan_readback PROPERTIES TIMEOUT 30 FAIL_REGULAR_EXPRESSION "Validation Error|VUID-")
  endif()
endif()
